cmake_minimum_required(VERSION 3.19)
project(Ww3Consumer LANGUAGES C Fortran)
find_package(WW3 CONFIG REQUIRED)
add_executable(point_io "${WW3_PREFIX}/share/ww3/tests/test_io_points_bin.F90")
target_link_libraries(point_io PRIVATE WW3::WW3)
configure_file("${WW3_PREFIX}/share/ww3/tests/data/ww3_grid.inp" ww3_grid.inp COPYONLY)
enable_testing()
add_test(NAME installed-ww3-grid COMMAND "${WW3_PREFIX}/bin/ww3_grid")
set_tests_properties(installed-ww3-grid PROPERTIES FIXTURES_SETUP grid TIMEOUT 120)
add_test(NAME installed-ww3-point-io COMMAND point_io)
set_tests_properties(installed-ww3-point-io PROPERTIES FIXTURES_REQUIRED grid TIMEOUT 120)
